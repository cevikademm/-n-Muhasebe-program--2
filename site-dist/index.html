void bubble_sort(int a[], size_t n) {
  bool swapped;
  do {
    swapped = false;
    for (size_t j = 0; j + 1 < n; ++j) {
      if (a[j] > a[j + 1]) {
        swap(&a[j], &a[j + 1]);
        swapped = true;
      }
    }
    --n;
  } while (swapped);
}
